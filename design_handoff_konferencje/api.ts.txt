import seed from './konferencje.seed.json';
import type { Konferencja } from './types';
// import { srtNaAkapity, txtNaAkapity } from '../lib/srt-txt';

export function getKonferencje(): Konferencja[] {
  return seed as Konferencja[];
}
export function getKonferencja(nr: number): Konferencja {
  return (seed as Konferencja[]).find((k) => k.nr === nr) ?? (seed as Konferencja[])[0];
}
